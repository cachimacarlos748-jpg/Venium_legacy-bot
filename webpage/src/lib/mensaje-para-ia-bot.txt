/**
 * Cliente Base44 para el bot de WhatsApp (Programa de Creadores de Legacy Store).
 * Reimplementa las llamadas del SDK en Node puro (sin window/localStorage).
 */
export const BASE44_CONFIG = {
  serverUrl: "https://base44.app",
  appId: "6a5b9606e1931edeb474236e",
  botEmail: process.env.LEGACY_BOT_EMAIL || "cachimacarlos748@gmail.com",
  botPassword: process.env.LEGACY_BOT_PASSWORD || "",
} as const;

const API = `${BASE44_CONFIG.serverUrl}/api`;
const APP = `/apps/${BASE44_CONFIG.appId}`;

export type CreatorVideoStatus = "pending"|"approved"|"counting"|"manual"|"completed"|"rejected";
export type RewardTier = "none"|"inicial"|"intermedio"|"pro";

export interface Creator { id:string; name:string; email:string; whatsapp:string; tiktok_handle?:string; code?:string; status:"pending"|"approved"|"rejected"; }
export interface CreatorVideo {
  id:string; creator_email:string; creator_name?:string; tiktok_url:string; game:string; game_id:string;
  status:CreatorVideoStatus; views_initial:number; views_current:number;
  reward_tier:RewardTier; reward_label?:string; manual_note?:string; view_check_error?:string;
  created_date?:string; check_attempts?:number;
}

export class Base44Client {
  private token: string | null = null;
  private expiry = 0;

  async login(): Promise<void> {
    const res = await fetch(`${API}${APP}/auth/login`, {
      method: "POST",
      headers: this.headers(),
      body: JSON.stringify({ email: BASE44_CONFIG.botEmail, password: BASE44_CONFIG.botPassword }),
    });
    if (!res.ok) throw new Error(`login ${res.status}: ${await res.text().catch(()=> "")}`);
    const d = await res.json() as { access_token?: string };
    if (!d.access_token) throw new Error("login sin access_token");
    this.token = d.access_token;
    this.expiry = Date.now() + 50*60*1000;
  }

  private headers(): Record<string,string> {
    const h: Record<string,string> = {
      "Content-Type":"application/json","Accept":"application/json",
      "X-App-Id": BASE44_CONFIG.appId,
    };
    if (this.token) h["Authorization"] = `Bearer ${this.token}`;
    return h;
  }

  private async req<T>(method:string, path:string, body?:unknown): Promise<T> {
    if (!this.token || Date.now() > this.expiry) await this.login();
    const doFetch = () => fetch(`${API}${path}`, {
      method,
      headers: this.headers(),
      body: body ? JSON.stringify(body) : undefined,
    });
    let res = await doFetch();
    if (res.status === 401) { await this.login(); res = await doFetch(); }
    if (!res.ok) throw new Error(`Base44 ${method} ${path} ${res.status}: ${await res.text().catch(()=> "")}`);
    if (res.status === 204) return undefined as T;
    return (await res.json()) as T;
  }

  // CRUD genérico
  filter<E>(entity:string, q:Record<string,unknown>, sort?:string): Promise<E[]> {
    const p = new URLSearchParams(); p.set("q", JSON.stringify(q)); if (sort) p.set("sort", sort);
    return this.req<E[]>("GET", `${APP}/entities/${entity}?${p}`);
  }
  get<E>(entity:string, id:string): Promise<E> { return this.req<E>("GET", `${APP}/entities/${entity}/${id}`); }
  create<E>(entity:string, data:Partial<E>): Promise<E> { return this.req<E>("POST", `${APP}/entities/${entity}`, data); }
  update<E>(entity:string, id:string, data:Partial<E>): Promise<E> { return this.req<E>("PUT", `${APP}/entities/${entity}/${id}`, data); }
  remove(entity:string, id:string): Promise<void> { return this.req<void>("DELETE", `${APP}/entities/${entity}/${id}`); }

  // Helpers del Programa de Creadores
  pendingVideos()                            { return this.filter<CreatorVideo>("CreatorVideo", { status: "pending" }, "created_date"); }
  creatorByEmail(email:string)               { return this.filter<Creator>("Creator", { email }); }
  approveVideo(id:string)                    { return this.update<CreatorVideo>("CreatorVideo", id, { status:"counting", views_initial:0, views_current:0, check_attempts:0 }); }
  rejectVideo(id:string, note:string)        { return this.update<CreatorVideo>("CreatorVideo", id, { status:"rejected", manual_note:note }); }
  finalizeVideo(id:string, views:number, tier:RewardTier, label:string) {
    return this.update<CreatorVideo>("CreatorVideo", id, { status:"completed", views_current:views, reward_tier:tier, reward_label:label });
  }
  markManual(id:string, err:string)          { return this.update<CreatorVideo>("CreatorVideo", id, { status:"manual", view_check_error:err }); }
  async setCreatorCode(email:string, code:string): Promise<Creator> {
    const [c] = await this.creatorByEmail(email); if (!c) throw new Error("creador no encontrado");
    return this.update<Creator>("Creator", c.id, { code, status:"approved" });
  }
  ping() { return this.req("GET", `${APP}/entities/User/me`).then(()=>true).catch(()=>false); }
}

export const base44 = new Base44Client();
